function procesarYGraficar(registros) { let sedesMap = {}; registros.forEach(item => { // Limpiar y obtener el nombre real de la sede let sedeRaw = String(item.SEDES || item.sedes || item.SEDE || item.sede || "").toUpperCase().trim(); // Omitir si la sede viene vacía, es un "0", o dice "SIN SEDE" if (!sedeRaw || sedeRaw === "0" || sedeRaw === "SIN SEDE" || sedeRaw === "NULL") { return; } let mod = String(item.MODULO || item.modulo || "").trim(); let tramites = parseInt(item.TOTAL_DE_TRAMITES || item.total_de_tramites || 0); let entregas = parseInt(item.ENTREGADAS || item.entregas || 0); let diasTrabajados = parseInt(item.DIAS_TRABAJADOS || item.dias_trabajados || 1); let disponibles = parseInt(item.DISPONIBLES_TOTAL || item.disponibles_total || 0); if (!sedesMap[sedeRaw]) { sedesMap[sedeRaw] = { modulo: mod, tramites: 0, entregas: 0, visitas: 0, pendientes: 0 }; } sedesMap[sedeRaw].tramites += tramites; sedesMap[sedeRaw].entregas += entregas; sedesMap[sedeRaw].visitas += (diasTrabajados > 0 ? diasTrabajados : 6); sedesMap[sedeRaw].pendientes += disponibles; }); let labels = Object.keys(sedesMap); let datosTramites = labels.map(s => sedesMap[s].tramites); let datosEntregas = labels.map(s => sedesMap[s].entregas); let datosVisitas = labels.map(s => sedesMap[s].visitas); // Actualizar Gráfica de Trámites vs Entregas const ctx1 = document.getElementById('graficaTramites').getContext('2d'); if (chartTramites) chartTramites.destroy(); chartTramites = new Chart(ctx1, { type: 'bar', data: { labels: labels.length > 0 ? labels : ['Sin datos válidos'], datasets: [ { label: 'Trámites', data: datosTramites.length > 0 ? datosTramites : [0], backgroundColor: 'rgba(13, 110, 253, 0.7)', borderRadius: 4 }, { label: 'Entregas', data: datosEntregas.length > 0 ? datosEntregas : [0], backgroundColor: 'rgba(25, 135, 84, 0.7)', borderRadius: 4 } ] }, options: { responsive: true, maintainAspectRatio: false, animation: { duration: 1000 }, scales: { y: { beginAtZero: true } } } }); // Actualizar Gráfica de Visitas const ctx2 = document.getElementById('graficaVisitas').getContext('2d'); if (chartVisitas) chartVisitas.destroy(); chartVisitas = new Chart(ctx2, { type: 'line', data: { labels: labels.length > 0 ? labels : ['Sin datos válidos'], datasets: [{ label: 'Días / Visitas a Sede', data: datosVisitas.length > 0 ? datosVisitas : [0], backgroundColor: 'rgba(220, 53, 69, 0.1)', borderColor: '#dc3545', borderWidth: 2, tension: 0.3, fill: true }] }, options: { responsive: true, maintainAspectRatio: false, animation: { duration: 1200 }, scales: { y: { beginAtZero: true } } } }); // Poblar Tablas de Control let tbodyVisitas = ''; let tbodyPendientes = ''; if (labels.length === 0) { tbodyVisitas = `No hay registros válidos para este filtro.`; tbodyPendientes = `No hay registros válidos para este filtro.`; } else { labels.forEach(sede => { let info = sedesMap[sede]; tbodyVisitas += `${info.modulo}${sede}${info.visitas}`; tbodyPendientes += `${sede}${info.tramites}${info.pendientes}`; }); } document.getElementById('tablaVisitasBody').innerHTML = tbodyVisitas; document.getElementById('tablaPendientesBody').innerHTML = tbodyPendientes; }
